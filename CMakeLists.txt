cmake_minimum_required(VERSION 3.21)
project(DiskCuratorNative VERSION 1.0.4 LANGUAGES CXX)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/diskcurator-version.txt" "${PROJECT_VERSION}\n")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Link the MSVC runtime statically so the release EXE works on a clean Windows 10/11
# installation without an additional Visual C++ Redistributable download.
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  # All project targets, including test executables, contain UTF-8 source.
  # Do not interpret UTF-8 Chinese fixture names using code page 936.
  add_compile_options(/utf-8)
endif()
option(DC_BUILD_TESTS "Build portable core tests" ON)
option(DC_BUILD_PERF_TOOLS "Build read-only performance assessment tool" OFF)
add_library(diskcurator_core STATIC src/core.cpp src/hash_index.cpp src/observations.cpp src/hash_actions.cpp src/scoped_analysis.cpp)
target_include_directories(diskcurator_core PUBLIC src)
if(WIN32)
  enable_language(RC)
  target_sources(diskcurator_core PRIVATE src/photo_windows.cpp)
  target_compile_definitions(diskcurator_core PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(diskcurator_core PUBLIC windowscodecs ole32 oleaut32 bcrypt)
  configure_file(src/app.rc "${CMAKE_CURRENT_BINARY_DIR}/app.rc" @ONLY)
  configure_file(src/app.manifest "${CMAKE_CURRENT_BINARY_DIR}/app.manifest" @ONLY)
  add_executable(DiskCurator WIN32 src/win_main.cpp "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
  target_include_directories(DiskCurator PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(DiskCurator PRIVATE diskcurator_core comctl32 shell32 shlwapi user32 gdi32 comdlg32 uuid)
  target_compile_definitions(DiskCurator PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MINGW)
    target_link_options(DiskCurator PRIVATE -municode)
  endif()
  if(MSVC)
    target_compile_options(diskcurator_core PRIVATE /W4 /EHsc)
    target_compile_options(DiskCurator PRIVATE /W4 /EHsc)
    # app.rc already embeds resource 1 (RT_MANIFEST) from src/app.manifest.
    # Disable MSVC linker's second automatically generated manifest, which
    # otherwise conflicts with the .rc resource (CVT1100 / LNK1123).
    # Keep this target-specific: test executables use normal MSVC manifests.
    target_link_options(DiskCurator PRIVATE /MANIFEST:NO)
  endif()
else()
  target_sources(diskcurator_core PRIVATE src/photo_portable.cpp)
endif()
add_executable(dc_cli src/cli.cpp)
target_compile_definitions(dc_cli PRIVATE DISKCURATOR_VERSION="${PROJECT_VERSION}")
target_link_libraries(dc_cli PRIVATE diskcurator_core)
if(MINGW)
  target_link_options(dc_cli PRIVATE -municode)
endif()
if(DC_BUILD_PERF_TOOLS)
  add_executable(dc_benchmark tools/perf_benchmark.cpp)
  target_link_libraries(dc_benchmark PRIVATE diskcurator_core)
  if(WIN32)
    target_link_libraries(dc_benchmark PRIVATE psapi)
  endif()
  if(MINGW)
    target_link_options(dc_benchmark PRIVATE -municode)
  endif()
endif()
if(DC_BUILD_TESTS)
  enable_testing()
  function(dc_test target source test_name)
    add_executable(${target} ${source})
    target_link_libraries(${target} PRIVATE diskcurator_core)
    # Legacy tests use assert. Keep the same assertion settings on every test,
    # including the release_assertions_active tripwire.
    if(MSVC)
      target_compile_options(${target} PRIVATE "/FI${CMAKE_CURRENT_SOURCE_DIR}/tests/enable_assertions.hpp")
    else()
      target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${test_name} COMMAND ${target})
  endfunction()
  dc_test(dc_tests tests/test_core.cpp core)
  dc_test(dc_recovery_tests tests/test_recovery.cpp recovery_fault_injection)
  dc_test(dc_stress_tests tests/test_stress.cpp directory_stress)
  dc_test(dc_merge_tests tests/test_merge.cpp merge_and_two_copy_retention)
  dc_test(dc_recommend_tests tests/test_recommend.cpp read_only_merge_recommendations)
  dc_test(dc_workflow_tests tests/test_workflow.cpp integrated_workspace_workflow)
  dc_test(dc_structure_tests tests/test_structure.cpp suggested_directory_structure)
  dc_test(dc_foundation_tests tests/test_foundation.cpp manual_foundation_and_directory_pair)
  dc_test(dc_hash_policy_tests tests/test_hash_policy.cpp complete_sha256_and_retention_policy)
  dc_test(dc_observation_tests tests/test_observations.cpp scan_observation_perspectives)
  dc_test(dc_scope_tests tests/test_scan_scope.cpp full_disk_and_manual_scope)
  dc_test(dc_scan_progress_tests tests/test_scan_progress.cpp scan_progress_and_streaming_hash)
  dc_test(dc_hash_actions_tests tests/test_hash_actions.cpp scan_hash_action_policy)
  dc_test(dc_scan_cancel_tests tests/test_scan_cancellation.cpp scan_rollup_and_verification_cancel)
  dc_test(dc_scoped_analysis_tests tests/test_scoped_analysis.cpp selected_directory_pair_scope)
  dc_test(dc_assert_sanity tests/test_assert_enabled.cpp release_assertions_active)
endif()
